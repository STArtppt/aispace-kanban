# 数据源配置样例。改名成 <你的源名>.yaml 放在同一个目录里就生效。
#
# 后缀是 .yaml.txt 而不是 .yaml，所以它**不会**被当成一个真的源 ——
# 看板和 db_ingest.py 只认 input/sources/*.yaml。
#
# 口令、token、完整连接串一律不要写进这份文件。这里只写「账号在哪个环境变量里」，
# 值写进工作空间根目录的 .env（.env 已在 .gitignore 里，不会入库）。

name: 仓库业务库              # 显示名。不写就用文件名
engine: postgresql            # 只支持 postgresql / mysql
host: 127.0.0.1
port: 5432
database: warehouse
schemas:                      # schema 白名单。不写 = PG 取 public，MySQL 取当前库
  - public
user_env: WAREHOUSE_DB_USER           # 账号在哪个环境变量里（不是账号本身）
password_env: WAREHOUSE_DB_PASSWORD   # 口令在哪个环境变量里（不是口令本身）
samples: 0                    # 每张表抽几行样例进 manifest。默认 0 = 不抽（真实数据落盘要显式决定）
max_rows: 5000                # 单次查询落盘的行数上限，超了截断并在 manifest 里写明
timeout: 15                   # 语句超时（秒）
client: driver                # driver（默认）或 cli（走 psql，只保证 schema 采集）
